# 知识库挂载策略设计

**版本**: v3.1
**日期**: 2025-11-01
**状态**: 补充设计 - 知识库挂载与权限控制

---

## 📋 设计概述

知识库采用**三级挂载策略**：租户级、组织级、用户级，实现细粒度的访问控制。用户查询时，系统根据挂载关系和权限计算可访问的知识库范围。

### 核心原则

1. **层级挂载**：知识库可挂载到租户、组织、用户三个层级
2. **权限继承**：用户可访问的知识库 = 租户级 ∪ 组织级 ∪ 用户级
3. **查询限制**：初始阶段限制一次查询最多4个知识库
4. **权限优先级**：用户级 > 组织级 > 租户级
5. **依赖子项目0**：组织结构、用户归属完全依赖子项目0

## 1. 三级挂载模型

### 1.1 挂载层级关系

```
租户 (Tenant)
├── 知识库挂载 (租户级)
│   └── KB-1, KB-2, KB-3  ← 租户下所有用户可访问
│
├── 组织A (Organization)
│   ├── 知识库挂载 (组织级)
│   │   └── KB-4, KB-5  ← 组织A的用户可访问
│   │
│   ├── 用户1 (User)
│   │   └── 知识库挂载 (用户级)
│   │       └── KB-6  ← 仅用户1可访问
│   │
│   └── 用户2 (User)
│       └── 知识库挂载 (用户级)
│           └── KB-7  ← 仅用户2可访问
│
└── 组织B (Organization)
    ├── 知识库挂载 (组织级)
    │   └── KB-8  ← 组织B的用户可访问
    │
    └── 用户3 (User)
        └── 知识库挂载 (用户级)
            └── KB-9  ← 仅用户3可访问
```

### 1.2 访问权限计算

**用户1的可访问知识库**:
```
租户级: [KB-1, KB-2, KB-3]
组织级: [KB-4, KB-5]  (来自组织A)
用户级: [KB-6]

可访问范围 = {KB-1, KB-2, KB-3, KB-4, KB-5, KB-6}
查询时最多选择: 4个
```

**用户3的可访问知识库**:
```
租户级: [KB-1, KB-2, KB-3]
组织级: [KB-8]  (来自组织B)
用户级: [KB-9]

可访问范围 = {KB-1, KB-2, KB-3, KB-8, KB-9}
查询时最多选择: 4个
```

## 2. 数据模型设计

### 2.1 知识库挂载表（扩展）

```sql
-- 知识库挂载表（支持三级挂载）⭐ 核心表
CREATE TABLE knowledge_base_mounts (
    id BIGSERIAL PRIMARY KEY,
    kb_id VARCHAR(50) NOT NULL REFERENCES knowledge_bases(id) ON DELETE CASCADE,

    -- 挂载层级（三选一）
    mount_type VARCHAR(20) NOT NULL CHECK (mount_type IN ('tenant', 'organization', 'user')),

    -- 挂载目标（根据mount_type使用不同字段）
    tenant_id VARCHAR(50),        -- 租户级挂载：租户ID（来自子项目0）
    organization_id VARCHAR(50),  -- 组织级挂载：组织ID（来自子项目0）
    user_id VARCHAR(50),          -- 用户级挂载：用户ID（来自子项目0）

    -- 挂载信息
    mounted_by VARCHAR(50) NOT NULL,  -- 挂载操作者（来自子项目0）
    mounted_at TIMESTAMP DEFAULT NOW(),

    -- 权限配置
    permissions JSONB DEFAULT '{
        "can_read": true,
        "can_write": false,
        "can_delete": false,
        "can_share": false
    }',

    -- 状态
    is_active BOOLEAN DEFAULT true,

    -- 约束：同一个KB不能重复挂载到同一个目标
    UNIQUE(kb_id, mount_type, tenant_id),
    UNIQUE(kb_id, mount_type, organization_id),
    UNIQUE(kb_id, mount_type, user_id),

    -- 检查约束：根据mount_type，对应字段必须非空
    CHECK (
        (mount_type = 'tenant' AND tenant_id IS NOT NULL AND organization_id IS NULL AND user_id IS NULL) OR
        (mount_type = 'organization' AND organization_id IS NOT NULL AND tenant_id IS NULL AND user_id IS NULL) OR
        (mount_type = 'user' AND user_id IS NOT NULL AND tenant_id IS NULL AND organization_id IS NULL)
    )
);

-- 索引
CREATE INDEX idx_mount_tenant ON knowledge_base_mounts(tenant_id, is_active) WHERE mount_type = 'tenant';
CREATE INDEX idx_mount_org ON knowledge_base_mounts(organization_id, is_active) WHERE mount_type = 'organization';
CREATE INDEX idx_mount_user ON knowledge_base_mounts(user_id, is_active) WHERE mount_type = 'user';
CREATE INDEX idx_mount_kb ON knowledge_base_mounts(kb_id, mount_type, is_active);
```

## 3. 权限计算逻辑

### 3.1 获取用户可访问的知识库

```go
package kbservice

// 获取用户可访问的所有知识库 ⭐ 核心方法
func (s *KnowledgeBaseService) GetUserAccessibleKBs(ctx context.Context, userID string) ([]string, error) {
    // 1. 从子项目0获取用户信息
    user, err := s.userCenterClient.GetUser(userID)
    if err != nil {
        return nil, err
    }

    var kbIDs []string

    // 2. 获取租户级知识库
    var tenantKBs []string
    err = s.db.Raw(`
        SELECT DISTINCT kb_id
        FROM knowledge_base_mounts
        WHERE mount_type = 'tenant'
        AND tenant_id = ?
        AND is_active = true
    `, user.TenantID).Scan(&tenantKBs).Error
    if err != nil {
        return nil, err
    }
    kbIDs = append(kbIDs, tenantKBs...)

    // 3. 获取组织级知识库
    if user.OrganizationID != "" {
        var orgKBs []string
        err = s.db.Raw(`
            SELECT DISTINCT kb_id
            FROM knowledge_base_mounts
            WHERE mount_type = 'organization'
            AND organization_id = ?
            AND is_active = true
        `, user.OrganizationID).Scan(&orgKBs).Error
        if err != nil {
            return nil, err
        }
        kbIDs = append(kbIDs, orgKBs...)
    }

    // 4. 获取用户级知识库
    var userKBs []string
    err = s.db.Raw(`
        SELECT DISTINCT kb_id
        FROM knowledge_base_mounts
        WHERE mount_type = 'user'
        AND user_id = ?
        AND is_active = true
    `, userID).Scan(&userKBs).Error
    if err != nil {
        return nil, err
    }
    kbIDs = append(kbIDs, userKBs...)

    // 5. 去重
    kbIDs = uniqueStrings(kbIDs)

    return kbIDs, nil
}

// 获取用户可访问的知识库（带缓存）
func (s *KnowledgeBaseService) GetUserAccessibleKBsWithCache(ctx context.Context, userID string) ([]string, error) {
    // 1. 尝试从Redis获取
    cacheKey := fmt.Sprintf("user_accessible_kbs:%s", userID)
    if cached, err := s.redis.Get(cacheKey).Result(); err == nil {
        var kbIDs []string
        json.Unmarshal([]byte(cached), &kbIDs)
        return kbIDs, nil
    }

    // 2. 从数据库查询
    kbIDs, err := s.GetUserAccessibleKBs(ctx, userID)
    if err != nil {
        return nil, err
    }

    // 3. 缓存（TTL: 5分钟）
    data, _ := json.Marshal(kbIDs)
    s.redis.Set(cacheKey, data, 5*time.Minute)

    return kbIDs, nil
}
```

```go
    userID := ctx.Value("user_id").(string)

    // 1. 获取用户可访问的所有知识库
    accessibleKBs, err := s.kbService.GetUserAccessibleKBsWithCache(ctx, userID)
    if err != nil {
        return nil, err
    }

    // 2. 应用限制：最多4个知识库 ⭐
    var selectedKBs []string
    if len(accessibleKBs) <= 4 {
        selectedKBs = accessibleKBs
    } else {
        // 策略1: 用户指定（如果提供了kb_ids参数）
        if len(req.KBIDs) > 0 {
            // 验证用户指定的知识库是否在可访问范围内
            for _, kbID := range req.KBIDs {
                if contains(accessibleKBs, kbID) {
                    selectedKBs = append(selectedKBs, kbID)
                    if len(selectedKBs) >= 4 {
                        break
                    }
                }
            }
        } else {
            // 策略2: 智能选择（根据相关性、使用频率等）

    return s.searchInKBs(ctx, selectedKBs, req.Query, req.TopK)
```

## 5. 权限验证

### 5.1 挂载权限验证

```go
// 验证用户是否有权限挂载知识库到目标
func (s *MountService) CanMount(ctx context.Context, req MountRequest) error {
    userID := ctx.Value("user_id").(string)

    switch req.MountType {
    case "tenant":
        // 租户级挂载：需要租户管理员权限
        hasPermission, _ := s.userCenterClient.CheckPermission(
            userID, "tenant", "manage_kbs",
        )
        if !hasPermission {
            return errors.New("需要租户管理员权限")
        }

    case "organization":
        // 组织级挂载：需要组织管理员权限
        hasPermission, _ := s.userCenterClient.CheckPermission(
            userID, fmt.Sprintf("organization:%s", req.OrganizationID), "manage_kbs",
        )
        if !hasPermission {
            return errors.New("需要组织管理员权限")
        }

    case "user":
        // 用户级挂载：需要是目标用户本人或管理员
        if req.UserID != userID {
            hasPermission, _ := s.userCenterClient.CheckPermission(
                userID, "tenant", "manage_user_kbs",
            )
            if !hasPermission {
                return errors.New("只能为自己挂载知识库或需要管理员权限")
            }
        }
    }

    return nil
}
```

```go
func (s *MountService) CheckMountQuota(ctx context.Context, req MountRequest) error {
    // 从子项目0获取租户配额
    quota, err := s.userCenterClient.GetTenantQuota(req.TenantID)
    if err != nil {
        return err
    }

    // 统计当前挂载数量
    var count int64
    s.db.Model(&KnowledgeBaseMount{}).
        Where("mount_type = ? AND is_active = true", req.MountType).
        Count(&count)

    // 验证配额
    switch req.MountType {
    case "tenant":
        if count >= quota.MaxTenantKBs {
            return errors.New("租户级知识库挂载已达上限")
        }
    case "organization":
        if count >= quota.MaxOrgKBs {
            return errors.New("组织级知识库挂载已达上限")
        }
    case "user":
        if count >= quota.MaxUserKBs {
            return errors.New("用户级知识库挂载已达上限")
        }
    }

    return nil
}
```
